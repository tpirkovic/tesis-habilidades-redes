% Tablas
\usepackage{booktabs}        % \toprule, \midrule, \bottomrule
\usepackage{array}
\usepackage{float}           % [H] para fijar figuras y tablas
% Quarto no siempre escribe [H] en el entorno table generado por kable: se fija
% por defecto para que la nota (\notatabla) quede siempre junto a su tabla o figura.
\floatplacement{table}{H}
\floatplacement{figure}{H}

% Ancho máximo de tablas: \ajustarancho{<tabular>} reduce la tabla al ancho de
% la caja de texto solo si lo excede; las tablas angostas quedan intactas.
\newsavebox{\cajatabla}
\newcommand{\ajustarancho}[1]{%
  \sbox{\cajatabla}{#1}%
  \ifdim\wd\cajatabla>\linewidth
    \resizebox{\linewidth}{!}{\usebox{\cajatabla}}%
  \else
    \usebox{\cajatabla}%
  \fi}

% Párrafos: sangría APA y sin espacio extra entre párrafos
\setlength{\parindent}{1.27cm}
\setlength{\parskip}{0pt}

% Títulos de tablas y figuras en formato APA: rótulo en negrita, título debajo
\usepackage{caption}
\captionsetup{
  labelfont = bf,
  labelsep  = newline,
  textfont  = it,
  justification = raggedright,
  singlelinecheck = false,
  font = {small}
}
\captionsetup[table]{position = top}
\captionsetup[figure]{position = top}

% Nota de tabla o figura: \notatabla{Texto de la nota.}
% La nota va en una caja que no se corta y no puede separarse de su tabla o
% figura: si no cabe, ambas pasan juntas a la página siguiente. Después de la
% nota queda una línea en blanco antes del párrafo siguiente.
\newcommand{\notatabla}[1]{%
  \par\nopagebreak[4]\vspace{0.3em}\nopagebreak[4]\noindent
  \begin{minipage}{\linewidth}\footnotesize\setlength{\parindent}{0pt}\textit{Nota.} #1\end{minipage}\par\vspace{\baselineskip}}

% Marcas de trabajo pendiente (visibles en el PDF, fáciles de buscar)
\usepackage{xcolor}
\newcommand{\pendiente}[1]{\textcolor{gray}{[Pendiente: #1]}}

% Letra pequeña en tablas. El hook va sobre el entorno table (no sobre tabular,
% que también usa \maketitle y rompe la portada).
\usepackage{etoolbox}
\AtBeginEnvironment{table}{\small\setstretch{1.1}}
\AtBeginEnvironment{longtable}{\small\setstretch{1.1}}

% Encabezado institucional de la primera página (escudo UC, instituto y
% magíster), a la manera de la plantilla UC de Overleaf. Se activa con
% \thispagestyle{primerauc} al comienzo de capitulos/00-resumen-dos-paginas.qmd.
\usepackage{fancyhdr}
\fancypagestyle{primerauc}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \fancyhead[L]{%
    \begin{picture}(0,0)
      \put(0,-14){\includegraphics[height=13mm]{img/logo-uc-escudo.png}}
      \put(34,4){\scriptsize\begin{tabular}{@{}l@{}}
        PONTIFICIA UNIVERSIDAD CATÓLICA DE CHILE\\
        INSTITUTO DE SOCIOLOGÍA\\
        MAGÍSTER EN SOCIOLOGÍA
      \end{tabular}}
    \end{picture}}%
  \fancyfoot[C]{\thepage}%
}

% Espacio después de las hipótesis y del resumen (entorno quote), para que el
% párrafo siguiente no quede pegado.
\AfterEndEnvironment{quote}{\par\medskip}

% Más espacio entre columnas de las tablas
\setlength{\tabcolsep}{8pt}
